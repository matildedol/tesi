\section{Proofs}\label{app: proofs}
We first recall some basic results that are used throughout, and then give the proofs of the results in the main text.

\begin{theorem}[Hoeffding's inequality]\label{app: hoeffding}
  Let $Z_1,\ldots,Z_n$ be independent real-valued random variables with $Z_i\in[a_i,b_i]$ almost surely for some $a_i,b_i\in\R$, and let $\mu = \frac1n\sum_{i=1}^n \expv[Z_i]$. Then, for every $\varepsilon>0$,
  \[
  \Pr\left[\frac1n\sum_{i=1}^n Z_i - \mu \ge \varepsilon\right] \le \exp\left(-\frac{2n^2\varepsilon^2}{\sum_{i=1}^n (b_i-a_i)^2}\right).
  \]
\end{theorem}

\begin{theorem}[Chebyshev's inequality]\label{app: chebyshev}
  Let $Z$ be a real-valued random variable with finite variance. Then, for every $\varepsilon>0$,
  \[
  \Pr\left[\vert Z - \expv[Z]\vert \ge \varepsilon\right] \le \frac{\Var(Z)}{\varepsilon^2}.
  \]
\end{theorem}

\begin{theorem}[Bernstein's inequality]\label{app: bernstein}
  Let $Z_1,\ldots,Z_n$ be independent real-valued random variables with $\expv[Z_i]=0$ and $Z_i\le M$ almost surely for some $M>0$, and let $v = \frac1n\sum_{i=1}^n \mathrm{Var}(Z_i)$. Then, for every $\varepsilon>0$,
  \[
  \Pr\left[\frac1n\sum_{i=1}^n Z_i \ge \varepsilon\right] \le \exp\left(-\frac{n\varepsilon^2}{2(v + M\varepsilon/3)}\right).
  \]
\end{theorem}

\begin{proposition}[Union bound]\label{app: unionbound}
  For any finite or countable collection of events $A_1, A_2, \ldots$,
  \[
  \Pr\left[\bigcup_i A_i\right] \le \sum_i \Pr[A_i].
  \]
\end{proposition}

\begin{theorem}[Hoeffding's lemma]\label{app: hoeffdinglemma}
  Let $Z$ be a real-valued random variable with $\expv[Z]=0$ and $\vert Z\vert\le a$ almost surely for some $a\in\R$. Then, for every $t\in\R$,
  \[
  \expv\left[e^{tZ}\right] \le e^{t^2a^2/2}.
  \]
\end{theorem}

\begin{proposition}[Tail integration]\label{app: tail int}
  For any non-negative random variable $Z$,
  \[
  \expv[Z] = \int_0^\infty \Pr[Z\ge \varepsilon] \de \varepsilon.
  \]
\end{proposition}

\begin{proposition}[Gaussian tail bound]\label{app: gtailbound}
  For every $c>0$ and $x_0>0$,
  \[
  \int_{x_0}^\infty e^{-cx^2}\de x \le \frac{1}{2cx_0}e^{-cx_0^2}.
  \]
\end{proposition}

\begin{proof}[Proof of Proposition~\ref{prop: suff assump 2.2}] Let $\ell:A\to\R^+$ satisfy Assumption~\ref{ass: lipsch loss} with constants $C$ and $C'$. Then
  \begin{align*}
            \ell(a) &= \vert \ell(a)\vert \\
            &= \vert \ell(a)-\ell(0)+\ell(0)\vert \\
            &\le \vert \ell(a)-\ell(0)\vert + \vert \ell(0)\vert \\
            &\le C\vert a \vert + C^\prime \\
            &\le CB + C^\prime \eqqcolon M
  \end{align*}
\end{proof} 

\begin{proof}[Proof of Theorem~\ref{thm: unionbound}]
We prove the lower bound first. Since $\hat h_D \in \argmin_{h\in\hyp} \hat R_D(h)$, we have
\[
\hat R_D(\hat h_D) \le \hat R_D(h) \qquad \forall h\in\hyp,
\]
and taking expectations
\[
\expv_D[\hat R_D(\hat h_D)] \le \expv_D[\hat R_D(h)] \qquad \forall h\in\hyp.
\]
For a fixed $h\in\hyp$, which does not depend on $D$, the empirical risk is an unbiased estimate of the true risk, since by linearity and because the $(X_i,Y_i)$ are distributed as $(X,Y)$,
\[
\expv_D[\hat R_D(h)] = \frac1n\sum_{i=1}^n \expv\left[\ell(h(X_i),Y_i)\right] = R(h).
\]
Hence
\[
\expv_D[\hat R_D(\hat h_D)] \le R(h) \qquad \forall h\in\hyp.
\]
Now take $S$ a realization of $D$ and consider in particular $h=\hat h_{S}\in\hyp$,
\[
\expv_D[\hat R_D(\hat h_D)] \le R(\hat h_{S}).
\]
Since this holds for every realization $S$ of $D$, we have $\expv_D[\hat R_D(\hat h_D)] \le R(\hat h_D)$ almost surely and, taking expectations,
\[
\expv_D[\hat R_D(\hat h_D)] \le \expv_D[R(\hat h_D)].
\]

Next, we prove the upper bound. Fix $h\in\hyp$ and let $Z_i \coloneq -\ell(h(X_i),Y_i)$, $i=1,\ldots,n$, so that
\[
\frac1n\sum_{i=1}^n Z_i - \frac1n\sum_{i=1}^n \expv[Z_i] = (R - \hat R_D)(h).
\]
We want to apply Hoeffding's inequality (Theorem~\ref{app: hoeffding}). The $Z_i$ are independent, since the $(X_i,Y_i)$ are, and $Z_i\in[-M,0]$ by non-negativity of the loss function and Proposition~\ref{prop: suff assump 2.2}. Hence, with $a_i = -M$ and $b_i = 0$, we obtain
\begin{equation}\label{app/eq: hoeffding fixed h}
\Pr\left[(R - \hat R_D)(h) \ge \varepsilon\right] \le e^{-2n\varepsilon^2/M^2}.
\end{equation}
Now, to obtain a bound at $\hat h_D$, which depends on $D$, notice that if $(R-\hat R_D)(\hat h_D)\ge\varepsilon$ then $(R - \hat R_D)(h)\ge\varepsilon$ for some $h\in\hyp$. Therefore, by the union bound (Proposition~\ref{app: unionbound}) and~(\ref{app/eq: hoeffding fixed h}),
\begin{align}\label{app/eq: unionbound prob}
\Pr\left[(R-\hat R_D)(\hat h_D) \ge \varepsilon\right] &\le \Pr\left[\bigcup_{h\in\hyp}\left\{(R - \hat R_D)(h)\ge\varepsilon\right\}\right] \notag \\
&\le \sum_{h\in\hyp} \Pr\left[(R - \hat R_D)(h)\ge \varepsilon\right] \notag \\
&\le \card e^{-2n\varepsilon^2/M^2}.
\end{align}

To obtain a bound in expected value from~(\ref{app/eq: unionbound prob}), we tail integrate. Let $X\coloneq(R-\hat R_D)(\hat h_D)$ and $X^+\coloneq\max\{X,0\}$, which is non-negative, so that by Proposition~\ref{app: tail int}
\[
\expv_D[X] \le \expv_D[X^+] = \int_0^\infty \Pr[X^+\ge\varepsilon]\de\varepsilon = \int_0^\infty \Pr[X\ge\varepsilon]\de\varepsilon.
\]
Let $\alpha\coloneq\card$ and $c\coloneq 2n/M^2$, so that~(\ref{app/eq: unionbound prob}) reads $\Pr[X\ge\varepsilon]\le\alpha e^{-c\varepsilon^2}$. This bound is good for large values of $\varepsilon$, but for small $\varepsilon$ it exceeds $1$, hence we use instead
\[
\Pr[X\ge\varepsilon] \le \min\{1,\alpha e^{-c\varepsilon^2}\}.
\]
We split the integral at the threshold $\varepsilon_0$ where $\alpha e^{-c\varepsilon_0^2} = 1$, i.e.\ $\varepsilon_0 = \sqrt{\log\alpha/c}$. Using the Gaussian tail bound (Proposition~\ref{app: gtailbound}) for the second integral, we obtain
\[
\expv_D[X] \le \int_0^{\varepsilon_0} 1\de\varepsilon + \int_{\varepsilon_0}^\infty \alpha e^{-c\varepsilon^2}\de\varepsilon
\le \varepsilon_0 + \frac{\alpha}{2c\varepsilon_0}e^{-c\varepsilon_0^2}
= \sqrt{\frac{\log\alpha}{c}} + \frac{1}{2\sqrt{c\log\alpha}},
\]
where in the last equality we used $\alpha e^{-c\varepsilon_0^2}=1$. Finally, assuming $\alpha\ge e^{1/2}$, which holds as soon as $\card\ge 2$, we have $\frac{1}{2\sqrt{c\log\alpha}}\le\sqrt{\frac{\log\alpha}{c}}$, and hence
\[
\expv_D[X] \le 2\sqrt{\frac{\log\alpha}{c}} = 2M\sqrt{\frac{\log\card}{2n}}.
\]
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm: rad bound}]
Let $Z\coloneq(X,Y)$ and $Z_i\coloneq(X_i,Y_i)$, so that
\[
R(h) = \expv[\ell_h(Z)] \qquad \text{and} \qquad \hat R_D(h) = \frac1n\sum_{i=1}^n \ell_h(Z_i).
\]
The key ingredient of the proof is to introduce an independent copy $D'=(Z_1',\ldots,Z_n')$ of the dataset $D$, and to study the random variables $\ell_h(Z_i')-\ell_h(Z_i)$, which have a symmetric distribution ({\it symmetrization trick}). Since the $Z_i'$ are distributed as $Z$ and are independent of $D$,
\[
R(h) = \expv_{D'}\left[\frac1n\sum_{i=1}^n \ell_h(Z_i')\right] = \expv_{D'}\left[\frac1n\sum_{i=1}^n \ell_h(Z_i')\,\Big\vert\, D\right].
\]
Then,
\[
\expv_D\left[\sup_{h\in\hyp}(R-\hat R_D)(h)\right] = \expv_D\left[\sup_{h\in\hyp} \expv_{D'}\left[\frac1n\sum_{i=1}^n \left(\ell_h(Z_i')-\ell_h(Z_i)\right)\,\Big\vert\, D\right]\right].
\]
Since the supremum of expectations is at most the expectation of the supremum, and by the law of total expectation, we obtain
\begin{equation}\label{app/eq: sym step1}
\expv_D\left[\sup_{h\in\hyp}(R-\hat R_D)(h)\right] \le \expv_{D,D'}\left[\sup_{h\in\hyp}\frac1n\sum_{i=1}^n \left(\ell_h(Z_i')-\ell_h(Z_i)\right)\right].
\end{equation}
Now we apply the symmetrization trick. Let $\hyp = \{h_1,\ldots,h_m\}$, with $m = \card$. For each $i=1,\ldots,n$, since $Z_i$ and $Z_i'$ are independent and identically distributed, swapping them does not change the joint distribution, i.e.
\[
\left(\ell_{h_j}(Z_i')-\ell_{h_j}(Z_i)\right)_{j=1}^m \overset{d}{=} \left(\ell_{h_j}(Z_i)-\ell_{h_j}(Z_i')\right)_{j=1}^m.
\]
This means that, for any fixed $e=(e_1,\ldots,e_n)\in\{-1,1\}^n$,
\[
\left(\ell_{h_j}(Z_i')-\ell_{h_j}(Z_i)\right)_{j=1}^m \overset{d}{=} \left(e_i\left(\ell_{h_j}(Z_i')-\ell_{h_j}(Z_i)\right)\right)_{j=1}^m.
\]
Since the pairs $(Z_i,Z_i')$ are independent across $i$, we have
\[
\left(\frac1n\sum_{i=1}^n \left(\ell_{h_j}(Z_i')-\ell_{h_j}(Z_i)\right)\right)_{j=1}^m \overset{d}{=} \left(\frac1n\sum_{i=1}^n e_i\left(\ell_{h_j}(Z_i')-\ell_{h_j}(Z_i)\right)\right)_{j=1}^m,
\]
and, since the supremum over $\hyp$ is a function of this vector,
\[
\sup_{h\in\hyp}\frac1n\sum_{i=1}^n \left(\ell_h(Z_i')-\ell_h(Z_i)\right) \overset{d}{=} \sup_{h\in\hyp}\frac1n\sum_{i=1}^n e_i\left(\ell_h(Z_i')-\ell_h(Z_i)\right).
\]
Hence, the expected value of the left-hand side is equal to that of the right-hand side for every $e\in\{-1,1\}^n$. Averaging over $e$ with the distribution of $\xi=(\xi_1,\ldots,\xi_n)$, which is independent of $D$ and $D'$, we obtain
\[
\expv_{D,D'}\left[\sup_{h\in\hyp}\frac1n\sum_{i=1}^n \left(\ell_h(Z_i')-\ell_h(Z_i)\right)\right] = \expv_{\xi,D,D'}\left[\sup_{h\in\hyp}\frac1n\sum_{i=1}^n \xi_i\left(\ell_h(Z_i')-\ell_h(Z_i)\right)\right].
\]
Plugging this into~(\ref{app/eq: sym step1}) and bounding the supremum of a sum with the sum of the suprema,
\begin{align*}
\expv_D\left[\sup_{h\in\hyp}(R-\hat R_D)(h)\right] &\le \expv_{\xi,D'}\left[\sup_{h\in\hyp}\frac1n\sum_{i=1}^n \xi_i\ell_h(Z_i')\right] + \expv_{\xi,D}\left[\sup_{h\in\hyp}\frac1n\sum_{i=1}^n (-\xi_i)\ell_h(Z_i)\right] \\
&= 2\mathcal R_n(\LL),
\end{align*}
because $D'$ has the same distribution as $D$, and $-\xi$ has the same distribution as $\xi$ and is independent of $D$.
\end{proof}

\begin{proof}[Proof of Lemma~\ref{lemma: massart}]
Let $Z_i\coloneq(X_i,Y_i)$. For any $t>0$,
\begin{align}
\exp\left(t\,\expv_{\xi,D}\left[\sup_{h\in\hyp}\sum_{i=1}^n \xi_i\ell_h(Z_i)\right]\right)
&\le \expv_{\xi,D}\left[\exp\left(t\sup_{h\in\hyp}\sum_{i=1}^n \xi_i\ell_h(Z_i)\right)\right] \notag\\
&= \expv_{\xi,D}\left[\sup_{h\in\hyp}\exp\left(t\sum_{i=1}^n \xi_i\ell_h(Z_i)\right)\right] \notag\\
&\le \expv_{\xi,D}\left[\sum_{h\in\hyp}\exp\left(t\sum_{i=1}^n \xi_i\ell_h(Z_i)\right)\right] \notag\\
&= \expv_D\left[\sum_{h\in\hyp}\prod_{i=1}^n \expv_{\xi_i}\left[\exp\left(t\xi_i\ell_h(Z_i)\right)\right]\right], \label{app/eq: massart start}
\end{align}
where we use Jensen's inequality for the convex function $\exp$ in the first inequality, that $\exp$ is increasing in the first equality, we bound the supremum with the sum (all terms are positive) in the second inequality, and we use the independence of the $\xi_i$'s, and their independence of $D$, in the last equality.

Now we apply Hoeffding's lemma (Theorem~\ref{app: hoeffdinglemma}) to each factor, conditionally on $D$. We can do so because, for fixed $Z_i$, the random variable $\xi_i\ell_h(Z_i)$ is centered, since $\expv[\xi_i]=0$, and it satisfies $\vert\xi_i\ell_h(Z_i)\vert = \ell_h(Z_i)\le M$ by non-negativity of the loss and Proposition~\ref{prop: suff assump 2.2}. We obtain
\[
\expv_{\xi_i}\left[\exp\left(t\xi_i\ell_h(Z_i)\right)\right] \le e^{t^2M^2/2}.
\]
Plugging this into~(\ref{app/eq: massart start}),
\[
\exp\left(t\,\expv_{\xi,D}\left[\sup_{h\in\hyp}\sum_{i=1}^n \xi_i\ell_h(Z_i)\right]\right) \le \sum_{h\in\hyp}\prod_{i=1}^n e^{t^2M^2/2} = \card\, e^{nt^2M^2/2}.
\]
Taking the logarithm of both sides,
\[
t\,\expv_{\xi,D}\left[\sup_{h\in\hyp}\sum_{i=1}^n \xi_i\ell_h(Z_i)\right] \le \log\card + \frac{nt^2M^2}{2},
\]
and dividing by $tn>0$,
\begin{equation}\label{app/eq: massart opt}
\mathcal R_n(\LL) \le \frac{\log\card}{tn} + \frac{tM^2}{2}.
\end{equation}
Since~(\ref{app/eq: massart opt}) holds for every $t>0$, we look for the tightest bound. Setting the derivative in $t$ to zero,
\[
-\frac{\log\card}{t^2n} + \frac{M^2}{2} = 0 \quad\Longrightarrow\quad t^\star = \sqrt{\frac{2\log\card}{nM^2}},
\]
and the second derivative $\frac{2\log\card}{t^3n}\ge0$, hence $t^\star$ is a minimizer (if $\card = 1$, the bound follows by letting $t\to\infty$). Plugging $t^\star$ into~(\ref{app/eq: massart opt}) yields
\[
\mathcal R_n(\LL) \le M\sqrt{\frac{2\log\card}{n}}.
\]
\end{proof}

\begin{proof}[Proof of Lemma~\ref{lemma: massart loss v}]
Let $Z_i\coloneq(X_i,Y_i)$. Since $\xi$ is independent of $D$, by the tower law we can write
\[
\mathcal R_n(\LL) = \expv_D\left[\expv_\xi\left[\sup_{h\in\hyp}\frac1n\sum_{i=1}^n\xi_i\ell_h(Z_i)\right]\right],
\]
where the inner expectation is over $\xi$ only, with $D$ fixed. We bound the inner expectation for every realization $S$ of $D$, and then take the expectation over $D$. This is the reason why the expectation over $D$ remains outside of the bound: the number of loss vectors $N_D(\hyp)$ depends on $D$, hence, differently from the proof of Lemma~\ref{lemma: massart}, we cannot apply Jensen's inequality and optimize $t$ jointly over $\xi$ and $D$.

Fix $S=(z_1,\ldots,z_n)\in(\X\times\Y)^n$ and let $V_S\coloneq\{(\ell_h(z_1),\ldots,\ell_h(z_n)) : h\in\hyp\}$ be the set of loss vectors, so that $\vert V_S\vert = N_S(\hyp)$. The quantity $\sum_i\xi_i\ell_h(z_i)$ depends on $h$ only through its loss vector, hence
\[
\sup_{h\in\hyp}\sum_{i=1}^n\xi_i\ell_h(z_i) = \max_{v\in V_S}\sum_{i=1}^n\xi_i v_i.
\]
Then, for any $t>0$, proceeding as in the proof of Lemma~\ref{lemma: massart} with the expectation over $\xi$ only, we have
\begin{align*}
\exp\left(t\,\expv_\xi\left[\max_{v\in V_S}\sum_{i=1}^n\xi_i v_i\right]\right)
&\le \expv_\xi\left[\max_{v\in V_S}\exp\left(t\sum_{i=1}^n\xi_i v_i\right)\right]
\le \sum_{v\in V_S}\prod_{i=1}^n\expv_{\xi_i}\left[e^{t\xi_i v_i}\right] \\
&\le \sum_{v\in V_S}\prod_{i=1}^n e^{t^2M^2/2} = N_S(\hyp)\,e^{nt^2M^2/2},
\end{align*}
where we use Jensen's inequality, we bound the maximum with the sum and use the independence of the $\xi_i$'s, and we apply Hoeffding's lemma (Theorem~\ref{app: hoeffdinglemma}) to $\xi_iv_i$, which is centered and satisfies $\vert\xi_iv_i\vert = v_i\le M$. Taking the logarithm and dividing by $tn$,
\[
\expv_\xi\left[\sup_{h\in\hyp}\frac1n\sum_{i=1}^n\xi_i\ell_h(z_i)\right] \le \frac{\log N_S(\hyp)}{tn} + \frac{tM^2}{2}.
\]
Since $S$ is fixed, we can choose $t$ depending on $S$. Optimizing as in the proof of Lemma~\ref{lemma: massart}, with $t^\star = \sqrt{2\log N_S(\hyp)/(nM^2)}$, gives
\[
\expv_\xi\left[\sup_{h\in\hyp}\frac1n\sum_{i=1}^n\xi_i\ell_h(z_i)\right] \le M\sqrt{\frac{2\log N_S(\hyp)}{n}}
\]
(if $N_S(\hyp)=1$, the left-hand side is $0$ since $\expv[\xi_i]=0$). Since this holds for every $S\in(\X\times\Y)^n$, it holds for every realization of $D$, i.e.\ the inner expectation is bounded by $M\sqrt{2\log N_D(\hyp)/n}$ for every realization of $D$. Taking the expectation over $D$ yields the result.
\end{proof}

\begin{proof}[Proof of Lemma~\ref{lemma: counting}]
  We prove that $N_S(\hyp)\le \card$ and $N_S(\hyp)\le 2^n$ for every $S\in(\X\times\Y)^n$.
Fix $S=(z_1,\dots,z_n)$ and consider the map
\[
\Phi\colon \hyp \to {\{0,1\}}^n,\qquad
h \mapsto \bigl(\ell_h(z_1),\dots,\ell_h(z_n)\bigr),
\]
so that $N_S(\hyp)=|\Phi(\hyp)|$. Since the image of a set under any map
has cardinality at most that of the domain, $|\Phi(\hyp)|\le|\hyp|$.
Since $\Phi(\hyp)\subseteq\{0,1\}^n$, we also have
$|\Phi(\hyp)|\le|\{0,1\}^n|=2^n$. Because this holds for every realization $S$ of $D$, the claim for $N_D(\hyp)$ follows. 
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm: rademmassart bound}]
  Since $\hat h_S \in \hyp$ for any realization $S$ of $D$,
  \[
  \expv_D\left[(R-\hat R_D)(\hat h_D)\right] \le \expv_D\left[\sup_{h\in\hyp} (R-\hat R_D)(h)\right]
  \]
  The proof is concluded applying Theorem~\ref{thm: rad bound}, bounding the Rademacher complexity with Lemma~\ref{lemma: massart loss v}, and bounding $N_D(\hyp)$ with Lemma~\ref{lemma: counting}.
\end{proof}

\begin{proof}[Proof of Lemma~\ref{lemma: rel dev}]
Let $\LL=\{(x,y)\mapsto\ell(y,h(x)):h\in\hyp\}$,
where $\ell$ is the zero-one loss. Apply the first inequality of Theorem~5.1
of~\cite{boucheron2005theory}, with $\mathbb{S}_\mathcal{F}(X_1^{2n}) = N_{2n}(\hyp)$, yielding with probability at least $1-\delta$, simultaneously for all
$h\in\hyp$,
\[
R(h)-\widehat R_D(h)
\le
2\sqrt{\frac{R(h)\log(4N_{2n}(\hyp)/\delta)}{n}}.
\]
Set
\[
\delta_\varepsilon
=
4N_{2n}(\hyp)e^{-n\varepsilon^2/4}.
\]
If $\delta_\varepsilon<1$, applying the preceding inequality
with $\delta=\delta_\varepsilon$ gives
\[
\Pr\!\left[
\exists h\in\hyp:
R(h)-\widehat R_D(h)>\varepsilon\sqrt{R(h)}
\right]
\le
4N_{2n}(\hyp)e^{-n\varepsilon^2/4}.
\]
If $\delta_\varepsilon\ge1$, this probability bound holds
trivially. Finally, since $n\varepsilon^2\ge8$,
\[
4e^{-n\varepsilon^2/4}
\le
2e^{-n\varepsilon^2/16},
\]
which proves the claim.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm: bound emprisk}]
Let $\delta\in(0,1)$ and set
\[
u\coloneq\frac{\log(2N_{2n}(\hyp)/\delta)}{n} \qquad\text{and}\qquad \varepsilon\coloneq 4\sqrt{u},
\]
so that $2N_{2n}(\hyp)e^{-n\varepsilon^2/16} = \delta$. Since $N_{2n}(\hyp)\ge1$ and $\delta<1$, we have $n\varepsilon^2 = 16\log(2N_{2n}(\hyp)/\delta) \ge 16\log 2 \ge 8$, hence Lemma~\ref{lemma: rel dev} applies and gives
\[
\Pr\left[\exists h\in\hyp : (R-\hat R_D)(h) > \varepsilon\sqrt{R(h)}\right] \le \delta.
\]
Therefore, with probability at least $1-\delta$, we have $(R-\hat R_D)(h)\le\varepsilon\sqrt{R(h)}$ for all $h\in\hyp$, and in particular for $h=\hat h_D\in\hyp$.

On this event, we isolate $\hat R_D(\hat h_D)$. Let $r\coloneq R(\hat h_D)$ and $\hat r\coloneq\hat R_D(\hat h_D)$, so that $r-\varepsilon\sqrt r-\hat r\le0$. This is a quadratic inequality in $x\coloneq\sqrt r\ge0$, namely $x^2-\varepsilon x-\hat r\le0$, which implies that $x$ is at most the largest root,
\[
\sqrt r \le \frac{\varepsilon+\sqrt{\varepsilon^2+4\hat r}}{2}.
\]
Squaring both sides,
\[
r \le \frac{2\varepsilon^2 + 4\hat r + 2\varepsilon\sqrt{\varepsilon^2+4\hat r}}{4} = \hat r + \frac{\varepsilon^2}{2} + \frac{\varepsilon}{2}\sqrt{\varepsilon^2+4\hat r} \le \hat r + \varepsilon^2 + \varepsilon\sqrt{\hat r},
\]
where in the last inequality we used $\sqrt{a+b}\le\sqrt a+\sqrt b$. Hence, substituting $\varepsilon = 4\sqrt u$, with probability at least $1-\delta$,
\begin{align*}
(R-\hat R_D)(\hat h_D) &= r-\hat r \le \varepsilon\sqrt{\hat r} + \varepsilon^2 \\ &= 4\sqrt{\frac{\hat R_D(\hat h_D)\log(2N_{2n}(\hyp)/\delta)}{n}} + 16\frac{\log(2N_{2n}(\hyp)/\delta)}{n}.
\end{align*}
\end{proof}

\begin{proof}[Proof of Corollary~\ref{coroll: bound bestrisk}]
        Recall $h_0 = \argmin_{h\in\hyp} R(h)$, $R(h_0) = R^*_{\hyp}$ by definition of $R^*_\hyp$ and $\hat R_D(\hat h_D) \le \hat R_D(h_0)$ by definition of the ERM $\hat h_D$.
        Since $h_0$ is independent of the data, the variables $\ell(h_0(X_i),Y_i)$ are i.i.d.\ Bernoulli with mean $R^*_\hyp$ and variance $R^*_\hyp(1-R^*_\hyp) \le R^*_\hyp$. Bernstein inequality (Theorem~\ref{app: bernstein}) with $M=1$ gives, with probability at least $1-\delta/2$,
        \[
        \hat R_D(h_0) \le R^*_{\hyp} + \sqrt{\frac{2R^*_\hyp \log (2/\delta)}{n}} + \frac{2\log(2/\delta)}{3n}.
        \]
        Let $u\coloneq\log(4N_{2n}(\hyp)/\delta)/n$. Since $N_{2n}(\hyp)\ge 1$, $\log(2/\delta)/n\le u$. Thus
        \begin{equation}\label{eq: bound hatR h0}
        \hat R_D(\hat h_D) \le \hat R_D(h_0) \le R^*_\hyp + 2\sqrt{R^*_\hyp u} + u = {\left(\sqrt{R^*_\hyp} + \sqrt{u}\right)}^2.
        \end{equation}
        Theorem~\ref{thm: bound emprisk} at confidence $1-\delta/2$ gives, with probability at least $1-\delta/2$,
        \begin{equation}\label{eq: bo}
        (R-\hat R_D)(\hat h_D) \le 4\sqrt{\hat R_D(\hat h_D)\,u} + 16u.
        \end{equation}
        By a union bound, both events hold simultaneously with probability at least $1-\delta$. Hence, plugging~(\ref{eq: bound hatR h0}) in the RHS of~(\ref{eq: bo}), we obtain that with probability at least $1-\delta$,
        \begin{align*}
        (R-\hat R_D)(\hat h_D) &\le 4\sqrt u\left(\sqrt{R^*_\hyp} + \sqrt u\right) + 16u \\
        &= 4\sqrt{\frac{R^*_{\hyp}\log(4N_{2n}(\hyp)/\delta)}{n}} + 20\frac{\log(4N_{2n}(\hyp)/\delta)}{n}.
        \end{align*}
\end{proof}

\begin{proof}[Proof of Corollary~\ref{coroll: bound noiseandappr}]
  Substitute $R^*_\hyp = \sigma^2 + A_\hyp$ into Corollary~\ref{coroll: bound bestrisk}.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm: bound noiseandappr}]
    Let $\text{gap} = (R-\hat R_D)(\hat h_D)$ and $C = \sqrt{\frac{\log(4 N_{2n}(\hyp)/\delta)}{n}}$. Corollary~\ref{coroll: bound bestrisk} gives
\begin{align*}
&\Pr\left[\text{gap} \le 4\sqrt{R^*_\hyp}\,C + 20C^2\right] \ge 1-\delta \\
\implies &\Pr\left[\text{gap} > 4\sqrt{R^*_\hyp}\,C + 20C^2\right] \le \delta
\end{align*}
Set $\varepsilon = 4\sqrt{R^*_\hyp}\,C + 20C^2$. Let us now find $\delta(\varepsilon)$:
\[
20C^2 + 4\sqrt{R^*_\hyp}\,C - \varepsilon = 0 \implies C = \frac{-4\sqrt{R^*_\hyp} + \sqrt{16R^*_\hyp + 80\varepsilon}}{40} = \frac{\sqrt{R^*_\hyp + 5\varepsilon} - \sqrt{R^*_\hyp}}{10}
\]
So we have
\begin{align*}
\frac{\log(4N_{2n}(\hyp)/\delta)}{n} &= \left(\frac{\sqrt{R^*_\hyp + 5\varepsilon} - \sqrt{R^*_\hyp}}{10}\right)^2 \\
\frac{4N_{2n}(\hyp)}{\delta} &= \exp\left(n\left(\frac{\sqrt{R^*_\hyp + 5\varepsilon} - \sqrt{R^*_\hyp}}{10}\right)^2\right)
\end{align*}
and hence
\[
\delta(\varepsilon) = 4N_{2n}(\hyp)\exp\left(-\frac{n}{100}\left(\sqrt{R^*_\hyp + 5\varepsilon} - \sqrt{R^*_\hyp}\right)^2\right).
\]
Using that $\expv[X] = \int_0^\infty \Pr[X \ge \varepsilon]\,d\varepsilon$ for $X \ge 0$, applied to $\text{gap}^+ \coloneq \max\{\text{gap}, 0\}$,
\[
\expv_D[\text{gap}] \le \expv_D[\text{gap}^+] = \int_0^\infty \Pr[\text{gap} \ge \varepsilon]\,d\varepsilon \le \int_0^\infty \min\{1, \delta(\varepsilon)\}\,d\varepsilon.
\]
We now compute the integral by exchanging the roles of $\varepsilon$ and $\delta$. The map $\varepsilon\mapsto\delta(\varepsilon)$ is continuous and strictly decreasing on $[0,\infty)$, with $\delta(0) = 4N_{2n}(\hyp) > 1$ and $\delta(\varepsilon)\to0$ as $\varepsilon\to\infty$. Hence, for every $t\in(0,1)$ there is a unique $\varepsilon(t)>0$ such that $\delta(\varepsilon(t)) = t$, and it is given by $\varepsilon(t) = 4\sqrt{R^*_\hyp}\,C + 20C^2$ with $C = \sqrt{\log(4N_{2n}(\hyp)/t)/n}$, by construction of $\delta(\varepsilon)$. Moreover, for $t\in(0,1)$,
\[
\min\{1,\delta(\varepsilon)\} > t \iff \delta(\varepsilon) > t \iff \varepsilon < \varepsilon(t).
\]
Writing $\min\{1,\delta(\varepsilon)\} = \int_0^1 \mathbf 1\{t < \min\{1,\delta(\varepsilon)\}\}\,dt$ and exchanging the order of integration by Tonelli's theorem (the integrand is non-negative), we obtain
\[
\int_0^\infty \min\{1, \delta(\varepsilon)\}\,d\varepsilon = \int_0^1 \int_0^\infty \mathbf 1\{\varepsilon < \varepsilon(t)\}\,d\varepsilon\,dt = \int_0^1 \varepsilon(t)\,dt = \int_0^1 \varepsilon(\delta)\,d\delta.
\]
Hence, writing $\varepsilon(\delta) = 4\sqrt{R^*_\hyp}\,C + 20C^2$ explicitly,
\[
\expv_D[\text{gap}] \le \int_0^1 \varepsilon(\delta)\,d\delta = 4\sqrt{\frac{R^*_\hyp}{n}}\int_0^1 \sqrt{\log\frac{4N_{2n}(\hyp)}{\delta}}\,d\delta + \frac{20}{n}\int_0^1 \log\frac{4N_{2n}(\hyp)}{\delta}\,d\delta.
\]
The logarithm is the only part depending on $\delta$, and its integral is
\[
\int_0^1 \log\frac{4N_{2n}(\hyp)}{\delta}\,d\delta = \log(4N_{2n}(\hyp)) - \int_0^1 \log\delta\,d\delta = \log(4N_{2n}(\hyp)) + 1.
\]
Applying Jensen's inequality to the square root term,
\[
\int_0^1 \sqrt{\log\frac{4N_{2n}(\hyp)}{\delta}}\,d\delta \le \sqrt{\int_0^1 \log\frac{4N_{2n}(\hyp)}{\delta}\,d\delta} = \sqrt{\log(4N_{2n}(\hyp)) + 1},
\]
we obtain
\[
\expv_D[\text{gap}] \le 4\sqrt{\frac{R^*_\hyp\left(\log(4N_{2n}(\hyp)) + 1\right)}{n}} + 20\,\frac{\log(4N_{2n}(\hyp)) + 1}{n}.
\]

Asymptotically,
\[
\expv_D\left[(R-\hat R_D)(\hat h_D)\right] \le O\left(
\sqrt{\frac{(A_{\hyp}+\sigma^2)\,\log N_{2n}(\hyp)}{n}} + \frac{\log N_{2n}(\hyp)}{n} \right)
\]
because, at the numerator,
\[
\log (4 N_{2n}(\hyp)) + 1 = \log 4 + \log N_{2n}(\hyp) + 1 = \log N_{2n}(\hyp) + O(1)
\]
and for large $x$, 
\[
\log x + O(1) \le C \log x
\] 
for some constant $C>0$.
Apply Lemma~\ref{lemma: counting}, obtaining
\[
\expv_D\left[(R-\hat R_D)(\hat h_D)\right] \le \min\left\{ O\left(
\sqrt{\frac{(A_{\hyp}+\sigma^2)\log \card}{n}} + \frac{\log\card}{n} \right), O\left(\sqrt{A_{\hyp}+\sigma^2}+1 \right)\right\}
\]
\end{proof}

\begin{proof}[Proof of Lemma~\ref{lemma: growth}]
We prove the inequalities for $N_S(\hyp)$, for every $S\in(\X\times\Y)^n$. Since $N_D(\hyp)$ takes the value $N_S(\hyp)$ when $D$ takes the value $S$, they hold for $N_D(\hyp)$ for every realization of $D$.

Fix $S=((x_1,y_1),\ldots,(x_n,y_n))\in(\X\times\Y)^n$, and consider the maps
\[
\Psi\colon\hyp\to\Y^n,\quad h\mapsto\bigl(h(x_1),\ldots,h(x_n)\bigr),
\]
\[
\lambda\colon\Y^n\to\{0,1\}^n,\quad (\hat y_1,\ldots,\hat y_n)\mapsto\bigl(\ell(\hat y_1,y_1),\ldots,\ell(\hat y_n,y_n)\bigr).
\]
The map $\Phi\colon h\mapsto(\ell_h(x_1,y_1),\ldots,\ell_h(x_n,y_n))$ of the proof of Lemma~\ref{lemma: counting} factorizes as $\Phi=\lambda\circ\Psi$, hence $N_S(\hyp) = |\Phi(\hyp)| = |\lambda(\Psi(\hyp))|$. Moreover, by definition of growth function, $|\Psi(\hyp)|\le\Pi_n(\hyp)$.

\emph{$1\le N_S(\hyp)$.} Since $\hyp$ is nonempty, $\Phi(\hyp)$ is nonempty.

\emph{$N_S(\hyp)\le\Pi_n(\hyp)$.} Since the image of a set under any map has cardinality at most that of the set,
\[
N_S(\hyp) = |\lambda(\Psi(\hyp))| \le |\Psi(\hyp)| \le \Pi_n(\hyp).
\]

\emph{$\Pi_n(\hyp)\le\min\{\card,2^n\}$.} Let $x_1,\ldots,x_n\in\X$ and let $\Psi$ be the corresponding map. As above, $|\Psi(\hyp)|\le\card$, and $\Psi(\hyp)\subseteq\Y^n$ with $|\Y^n| = 2^n$, since $\Y=\{-1,1\}$. Taking the supremum over $x_1,\ldots,x_n$ gives $\Pi_n(\hyp)\le\card$ and $\Pi_n(\hyp)\le2^n$.

\emph{$\Pi_n(\hyp) = N_n(\hyp)$.} Taking the supremum over $S$ in $N_S(\hyp)\le\Pi_n(\hyp)$ gives $N_n(\hyp)\le\Pi_n(\hyp)$. For the converse, let $x_1,\ldots,x_n\in\X$, take any $y_1,\ldots,y_n\in\Y$, and let $S=((x_1,y_1),\ldots,(x_n,y_n))$. Since $\Y=\{-1,1\}$ and $\ell$ is the zero-one loss, $\hat y_i = y_i$ if $\ell(\hat y_i,y_i)=0$ and $\hat y_i = -y_i$ if $\ell(\hat y_i,y_i)=1$. Hence $\lambda$ is injective, and
\[
|\Psi(\hyp)| = |\lambda(\Psi(\hyp))| = N_S(\hyp) \le N_n(\hyp).
\]
Taking the supremum over $x_1,\ldots,x_n$ gives $\Pi_n(\hyp)\le N_n(\hyp)$.
\end{proof}

\[
y \leftarrow sign(h^*(x))b
\]
for $b\sim\mathrm{Bernoulli}(p)$. Notice that this noise flips the labels with a probability of $1-p$. 
While from the first experiments it seemed that the algorithm achieved a higher test error, we postulate that this may be due to the portion of misclassified inputs, rather than to the noise structure itself. Hence, fixing $p = 0.8$, we look for the value of $\sigma$ that achieves roughly the same portion of mislcassified inputs ($20\%$). Figure~\ref{app/fig: sigmastar parabola} shows that this happens for $\sigma = 5.5$, and Figures~\ref{app/fig: parabola bernoulli} and~\ref{app/fig: parabola gauss} show that the two noise structures then yield essentially the same train and test errors.

Finally, we investigate the afore-mentioned intuition about how easy it is for the algorithm to learn a simple function. We take $m=2$ (which corresponds to a parabola, in terms of classification pattern), sample uniformly from $\X$, and look for the training set size $n$ for which the test error gets within $0.01$ of the Bayes risk on average, at depth $2$ and $3$.

We start with little noise, $\sigma = 1$, for which the Bayes risk is $\approx 0.05$ (Figure~\ref{app/fig: simple function sigma1}). Here the algorithm learns the function quickly: with $n = 50$ points the test error is already $\approx 0.07$, within $0.02$ of the Bayes risk, and it gets within $0.01$ with $n = 200$ at depth $2$ and $n = 500$ at depth $3$.

With our noise level, $\sigma = 3.5$, the Bayes risk is $\approx 0.20$ and $n = 1000$ training points are needed on average for the test error to get within $0.01$ of it, both at depth $2$ and $3$ (Figure~\ref{app/fig: simple function}). Hence, although two splits are enough to represent the function, with this amount of noise the algorithm needs many points to place them correctly.
