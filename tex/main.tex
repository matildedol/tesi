\section{Mathematical framework}\label{sec: math}

We observe $x\in \X = \{\bar x_1, \ldots, \bar x_N\}\subset \R$, and $y\in \Y = \{-1,1\}$ and we would like to predict $y$ from $x$.
To do so, we are given a collection of observations like $x$ and $y$, i.e.\ our data. This allows to observe the relationship between $x$ and $y$ multiple times and hope to estimate it well.
% am i biased by causal inference nel parlare di relazione tra x e y? alla fine dobbiamo sempre trovare una relazione, però è forse misleading dato che è occulta in (most of) machine learning 

\begin{assumption}\label{1ass: randomdata}
    We assume our data is drawn from a probability distribution.
Therefore, we consider the sample space $\Omega$ and define the random variables 
\[\begin{gathered}
    X:\Omega \to \X, \quad \omega \mapsto x \qquad \text{and}\qquad Y:\Omega \to \Y,\quad \eta \mapsto y;\\
    (X,Y)\sim p,
\end{gathered}\]
with joint law $p$, which is a cathegorical probability distribution over  $\X\times\Y = \{\bar x_1,\ldots,\bar x_N\}\times\{-1,1\}$.
\end{assumption}

Since we want to obtain a collection of observations, we define an additional random variable.

\begin{definition}[Dataset]
    Given a size $n$, a sample space $\Omega$, and value spaces $\X,\Y$, a dataset $D$ is a random variable 
    \[
    \begin{gathered}
        D:\Omega^n \to \X^n\times\Y^n, \qquad D\coloneq {(X_i, Y_i)}_{i=1}^n, \\
     (\omega_1,\ldots,\omega_n)\mapsto ((x_1,y_1),\ldots,(x_n,y_n))\\
     D\sim p^n
        % D((\omega_1,\ldots,\omega_n),(\eta_1,\ldots,\eta_n)):=((X_1(\omega_1),Y_1(\eta_1)),\ldots,(X_n(\omega_n),Y_n(\eta_n))),\quad D\sim p^n
    \end{gathered}
    \]
    where the $X_i$s and $Y_i$s are i.i.d copies of $X,Y$, and $X,Y,p$ are as in Assumption~\ref{1ass: randomdata}. We indicate with the lower-case $d$ realizations of $D$, $d={\{(x_i,y_i)\}}_{i=1}^n$. Also, we use $\mathcal{D}$ to indicate the output space of $D$ in general, and treat $\mathcal D = \X^n\times\Y^n$ as a particular case. 
\end{definition}

Now, predicting $y$ from $x$ means we want to find the best function $f$ such that $f(x)$ is a good approximation of $y$. To do so, we consider an {\it\/hypotheses space} $\cH$, which is the set of all functions $f:\X\to\Y$ that satisfy some hypotheses we choose to make on the relationship between $x$ and $y$.
\begin{fact}
    Functions in $\cH$ are deterministic objects.
\end{fact}

In order to find the best predictor $f$, we use a {\it\/learning algorithm}, in the idea of learning our predictor from the data. 

\begin{definition}[Learning algorithm]\label{1def: learningalg}
A learning algorithm $A$ is a deterministic function
\[
A:\mathcal D \to \mathcal H, \qquad d \mapsto \hat f_d
\]
that given some data $d\in \mathcal{D}$, outputs a prediction function $\hat f_d\in \mathcal{H}$.
\end{definition}

\begin{remark}
    The hat notation $\hat f$ is used only to distinguish an element of $\cH$ that comes from a learning algorithm, or that depends on data, from a generic element of $\cH$, indicated as $f$. 
\end{remark}

Now, we are ready to define our central object, the predictor. 
\begin{definition}[Predictor]
    A predictor $\hat f_D$ is a random variable 
    \[
    \hat f_D : \Omega^n  \to \mathcal H, \qquad \hat f_D \coloneq A \circ D, \qquad {(\omega_i)}_{i=1}^n\mapsto d \mapsto \hat f_d
    \]
    that depends on some data $D$ and on a learning algorithm $A$. The source of randomness is the dataset $D$.  
\end{definition}

Now, we would like to have a way to {\it\/ evaluate} our predictors, in order to select the best one. To do so, we use a metric called {\it\/expected risk}. First, we take a loss function $\ell:\Y\times\Y\to\R^+$ that takes the preidction $f(x)$ and the true value $y$ and outputs a number corresponding to the error in the prediction. 

\begin{definition}[Expected risk]
     Given a predictor $f$ and a loss function $\ell$, the expected risk is a deterministic function $R: (\X\times\Y) \times \cH \to \R$,
    \begin{equation}
        R(f) \coloneq \sum_{i=1}^m \sum_{y_i\in\{-1,1\}}\ell(f(\bar x_i),y_i)p(\bar x_i, y_i).
    \end{equation}
\end{definition}
The issue with using $R(f)$ as a metric, however, is that we do not know the true distribution $p$ in general. Hence, we approximate the expected risk with a new, emprical metric, the {\it\/empirical risk}, motivated by the law of large numbers. 

\begin{definition}[Empirical risk]
Given a dataset $D$, a predictor $f$, and a loss function $\ell$, the empirical risk is
    \begin{equation}
        \hat R_D(f) \coloneq \frac{1}{n}\sum_{i=1}^n \ell(f(X_i),Y_i), \qquad D = (X_1,Y_1),\ldots,(X_n,Y_n).
    \end{equation}
    It is a random variable due to the randomness in $D$, in the sense that $\hat R_D \coloneq \hat R \circ D$ for a deterministic function $\hat R:\cH \times \mathcal{D}\to\R$.
\end{definition}

\begin{remark}
    Notice that we implicitly define the two metrics as operators applying the loss function $\ell$ to their inputs, but we could equivalently define them taking inputs from $\R$, the image space of $\ell$. 
\end{remark}
For a deterministic estimator $f\in\cH$, the following result holds.

\begin{proposition}[Unbiasedness of the empirical risk]\label{1prop: unbiased}
    Given a scalar $a \in \R$, the empirical risk of $a$ is an unbiased estimator of the expected risk,
    \begin{equation}
    \expv[\hat R_D(a)] = R(a).
\end{equation}
\begin{proof}
A proof is found in the appendix (Proposition~\ref{app: unbiased}).
\end{proof}
\end{proposition}

\section{Evaluating the bias}\label{sec: bias}

With the technology defined above, we define an estimator given a dataset $D$ and an hypotheses space $\cH$ as
\begin{equation}\label{1eq: argmin}
    \hat f_D \coloneq \argmin_{f\in\cH} \hat R_D(f).
\end{equation}
This means that among all candidates in $\cH$, we select the one that minimizes the empirical risk. Solving problem (\ref{1eq: argmin}) corresponds to {\it\/fitting} the estimator to the data $D$ or {\it\/training} the estimator on $D$. 

Subsequently, it is natural to look for a measure of performance of $\hat f_D$, for instance to choose the best one among a set of different hypotheses (e.g.\ different architectures of a neural network), or to provide a measure of reliability of our prediction. However, our estimator is now a random variable of the data. This makes the result in Proposition~\ref{1prop: unbiased} not valid anymore: the empirical risk computed on a dataset $D$ is a biased estimate of the true, expected risk of an estimator trained on $D$. This intuitively comes from the correlation between datapoints, as seen in Example~\ref{app: toy}. 
As a solution, classical statistical theory provides splitting the dataset into two parts $(D_{train},D_{test})=D$ to preserve unbiasedness. Usually, we fit the estimator to $D_{train}$ and compute $\hat R$ on $D_{test}$, to obtain $\hat R_{D_{test}}$ as an unbiased estimate of the risk of $\hat f_{D_{train}}$. 

We begin by asking the following question: how many splits of the dataset are really necessary in practice? To answer this, we want to study the bias arising from using the same dataset for training and evaluating a model. 

We make the following assumptions.

\begin{assumption}[Boundedness of loss function]\label{ass: bounded loss}
    Given a loss function $\ell:\Y\times\Y\to\R^+$ and a prediction function $f:\X\to\Y$, 
    $\exists M\in\R$ s.t.
    \[
    \ell (f(x), y) \le M \qquad \forall x\in\X, y\in\Y.
    \]
\end{assumption}

\begin{assumption}[Lipschitzness of loss function and bound at zero]\label{ass: lipsch loss}
Given a loss function $\ell:\Y\times\Y\to\R^+$ and a prediction function $f:\X\to\Y$, 
    \begin{enumerate}
        \item $\exists C>0$ s.t.
        \[ \forall a,a^\prime\in\R, \vert \ell(a)-\ell(a^\prime)\vert \le C\vert a-a^\prime\vert\]

        \item $\exists C^\prime >0$ s.t.
        \[ \vert \ell(0)\vert \le C^\prime
        \]
    \end{enumerate}
\end{assumption}

Let us make an example of a loss function that satisfies these assumptions.

\begin{example}[Logistic loss function]\label{ex: logistic}
    Let $\ell (a) = \log (1+e^{-a}), a \in \R$. Notice that to apply this to our setting we take the product $f(x)\cdot y$ and then apply $\ell$.
    We have
    \[
    \ell'(a) = -\frac{e^{-a}}{1+e^{-a}} = -\frac{1}{1+e^a}\quad \Rightarrow \quad \vert \ell'(a)\vert < 1 \qquad\forall a \in \R
    \]
    hence 
    \[
    C = \sup_{a\in\R} \vert \ell^\prime\vert \le 1.
    \]
    Also,
    \[
    \ell(0) = \log 2 = C^\prime.
    \]
\end{example}

\begin{proposition}\label{prop: suff assump 2.2}
    For $a\in A\subset\R, A\coloneqq \{a\in\R : \vert a \vert \le B\text{ for some constant } B \in \R\}$, Assumption~\ref{ass: lipsch loss} $\Rightarrow$ Assumption~\ref{ass: bounded loss}. 
    \begin{proof} For $a\in A$,
        \begin{align*}
            \ell(a) &= \vert \ell(a)\vert \\
            &= \vert \ell(a)-\ell(0)+\ell(0)\vert \\
            &\le \vert \ell(a)-\ell(0)\vert + \vert \ell(0)\vert \\
            &\le C\vert a \vert + C^\prime \\
            &\le CB + C^\prime \eqqcolon M
        \end{align*}
    \end{proof} 
\end{proposition}
Notice that we can always take a {\it\/truncation map} $T:\R\to[-1,1]$, defined as 
\begin{equation}
    T(a) = \begin{cases}
        -1 & a \le -1\\
        a & -1<a<1\\
        1 & a\ge 1,
    \end{cases}
\end{equation}
so that $T\circ f(x)$ is bounded for any $f$ and $x$. Hence, we consider $A = \{T\circ f(x) : f \in\cH, x\in\X\}$ and $\mathcal{F}\coloneq\{f \vert f:\X \to \R\}$, $\cH \subseteq \mathcal{F}, \cH = \{T\circ f \vert f \in \mathcal{F}\}$. Now, we are ready to state our main result. 

\begin{proposition}[Bounds on the bias]\label{prop: main bound}
Consider data $D: \Omega^n \to \X^n \times \Y^n$, a predictor $\hat f_D \in \cH$, and a loss function $\ell:\Y\times\Y\to\R^+$. Let $\ell$ be as in Assumption~\ref{ass: lipsch loss}, so that $\ell(\cdot) \le M$ for some $M\in\R$ by Proposition~\ref{prop: suff assump 2.2}. Then, the empirical and the expected risk of $\hat f_D$ satisfy the following bound
\begin{equation}\label{eq: mainbounds}
    0 \le \expv_D[ R(\hat f_D) - \hat R_D(\hat f_D)] \le 2M\sqrt{\frac{\log \vert \cH \vert}{2n}}.
\end{equation}
\begin{proof}
We prove the lower bound ($0\le\expv_D\ldots$) first. All the expectations are over the randomness of the data $D$, we drop the subscript to light notation. We have
\begin{equation}\label{ineq: minimizer1}
\hat R_D(\hat f_D) \le\hat R_D(f) \quad \forall f\in \cH,
\end{equation}
since $\hat f_D \coloneq \argmin_{f\in\cH} \hat R_D (f) $. 
Then, take expectations 
\begin{equation}
    \expv[\hat R_D(\hat f_D)] \le \expv[\hat R_D(f)] \quad \forall f\in \cH.
\end{equation}
Now, since the functions in $\cH$ are deterministic, by Proposition~\ref{1prop: unbiased} we have
\begin{equation}
    \expv[\hat R_D(f)] = R(f) \quad \forall f\in\cH,
\end{equation}
so that 
\begin{equation}\label{1eq: keybound}
    \expv[\hat R_D(\hat f_D)] \le R(f) \quad \forall f\in \cH.
\end{equation}
Now take $d$ a generic realization of $D$ and consider in particular $f=\hat f_d$,
\begin{equation}
    \expv[\hat R_D(\hat f_D)] \le R(\hat f_d) \quad \forall d\in \mathcal D.
\end{equation}
Since this is valid pointwise for any $d$ realization of $D$, we have
\begin{equation}
    \expv[\hat R_D(\hat f_D)] \le R(\hat f_D) \quad \text{a.s.}
\end{equation}
and, finally, taking expectations
\begin{equation}
        \expv[\hat R_D(\hat f_D)] \le \expv[R(\hat f_D)] \quad \text{a.s.}
\end{equation}
Next, we prove the upper bound ($\expv \le 2 \sqrt{}\ldots$). For any $f \in \cH$, $\expv [\hat R_D (f)]=R(f)$. Let $Z_i \coloneq \ell(f(x_i),y_i)$ and have
\[
\hat R_D(f) =\frac{1}{n}\sum_i Z_i 
\]
We now want to apply Hoeffding's inequality (Proposition~\ref{app: hoeffding} in the appendix). To do so, we need to satisfy 3 conditions:
\begin{enumerate}
    \item $Z_1, \ldots, Z_n$ i.i.d.;
    \item $\expv[Z_1] = \cdots = \expv[Z_n]$;
    \item $b_i \le Z_i \le a_i$ for some $a_1, \ldots, a_n \in \R$ and $b_1,\ldots,b_n\in\R$. 
\end{enumerate}

We have that $(i)$ is satisfied by ${Z_i}s$ being i.i.d.\ by construction. Condition $(ii)$ is satisfied since all the $(x_i,y_i)$ have the same distribution. $(iii)$ is satisfied with $b_1=\cdots=b_n=0$ and $a_1 = \cdots = a_n = M$ by non-negativity of the loss function and Proposition~\ref{prop: suff assump 2.2} with our definition of $\cH$. 
Now we can apply Hoeffding's. We obtain 
\begin{equation*}
    \Pr [\frac{1}{n}\sum_i Z_i - \mu \ge \varepsilon] \le e^{-2n\varepsilon^2/M^2}
\end{equation*}
and thus
\begin{equation}\label{ineq: applyhoeffding}
    \Pr [ R(f) - \hat R_D(f) \ge \varepsilon]\le e^{- 2n\varepsilon^2/M^2}
\end{equation}
Now, to obtain a bound for $\hat f_D$ we bound it with the union of the probabilities of the events for each $f\in\cH$. 
We have
\begin{equation}
    \Pr[R(\hat f_D)-\hat R_D(\hat f_D) \ge \varepsilon] \le \bigcup_{f \in \cH} \Pr[R(f)-\hat R_D(f) \ge \varepsilon]
\end{equation}
trivially, and also 
\begin{equation}
    \bigcup_{f \in \cH} \Pr[R(f)-\hat R_D(f) \ge \varepsilon] \le \sum_{f \in \cH} \Pr[R(f)-\hat R_D(f) \ge \varepsilon] \le \sum_{f \in \cH} e^{- 2n\varepsilon^2/M^2} = \vert \cH \vert  e^{- 2n\varepsilon^2/M^2}
\end{equation}
where the first inequality follows by a union bound and the second by Hoeffding. All in all, we obtain
\begin{equation}\label{eq: upper bound}
    \Pr[R(\hat f_D)-\hat R_D(\hat f_D) \ge \varepsilon] \le \vert \cH \vert  e^{- 2n\varepsilon^2/M^2}
\end{equation}

To obtain our initial inequality from this result on a probability, we use a technique called tail integration. The key ingredient is the following
\begin{equation}\label{eq: keying}
    \expv[X] = \int_0^\infty \Pr(X\ge \varepsilon) \de \varepsilon
\end{equation}
which is true for any non-negative random variable $X$, see Proposition~\ref{app: expvcmf} in the appendix. Then, we want to use our bound (\ref{eq: upper bound}) for the integrand. Let $a\coloneq \vert\cH\vert$ and $c\coloneq 2n/M^2$, so that
\[
\Pr (X\ge \varepsilon) \le a e^{-c\varepsilon^2}.
\] 
Notice that for high values of $\varepsilon$, the bound is a very good one, but for $\varepsilon \gg 1$ it is bad. Indeed, we can use a better bound:
\begin{equation}
    \Pr[X\ge \varepsilon] \le \min\{1, a  e^{-c\varepsilon^2}\}
\end{equation}
Therefore, it makes sense to split the integral in (\ref{eq: keying}) in two parts with a threshold $a  e^{-c\varepsilon_0^2} = 1$. This means splitting the integral at $\varepsilon_0 = \sqrt{\frac{\log a}{c}}$. We obtain 
\begin{align*}
\expv[X] &= \int_0^\infty \Pr(X\ge \varepsilon) \de \varepsilon \le \int_0^{\sqrt\frac{\log a}{c}} 1 \de \varepsilon + \int_{\sqrt\frac{\log a}{c}}^\infty ae^{-c\varepsilon^2}\de \varepsilon\\
&=\sqrt\frac{\log a}{c} + a\frac{1}{2\sqrt{c\log a}}e^{-c\varepsilon^2} = \sqrt\frac{\log a}{c} + \frac{1}{2\sqrt{c\log a}}
\end{align*}
where the second integral is solved with a standard Gaussian tail bound (see Proposition~\ref{app: gtailbound}), and in the last equality we substituted $a = 1/e^{-c\varepsilon^2}$. Finally, we can use a looser but simpler bound assuming  $a\ge e^{1/2}$, which is sensible since $a=\vert\cH\vert $ and $e^{1/2}<2$,
\begin{equation}\label{eq}
    \expv[X] \le 2\sqrt\frac{\log a}{c}
\end{equation}
which yields the result for $X = R(\hat f_D)-\hat R(\hat f_D)$, $a = \vert \cH\vert$, and $c=2n/M^2$. 
\end{proof}
\end{proposition}

\begin{example}[Bound for the logistic loss function]
Let us see how this bound looks like for a logistic loss function. By our definition of $\cH$ via the truncation map, we have $B=1$; then, $C = 1$ and $C^\prime = \log2$ by Example~\ref{ex: logistic}. Hence,
\begin{equation}
        0 \le \expv_D[ R(\hat f_D) - \hat R_D(\hat f_D)] \le (2+2\log 2)\sqrt{\frac{\log \vert \cH \vert}{2n}}.
\end{equation}
\end{example}

\subsection{Rademacher Complexity and Symmetrization Trick}

In this section, we present a different bound from the one in Proposition~\ref{prop: main bound}, based on the so-called {\it\/Rademacher complexity} and on a {\it\/symmetrization trick}.
% We follow~\cite{bach2024learning}.

\begin{definition}[Rademacher complexity]\label{def: rademacher}
    Given a data size $n$, a dataset $D = {(X_i, Y_i)}_{i=1}^n$, a loss function $\ell:\Y\to\Y$, and an hypotheses space $\cH=\{f \vert f:\X\to\Y\}$, let $\varepsilon$ be a $n$-dimensional vector of i.i.d. Rademacher random variables (i.e.\ taking values $1$ and $-1$ with equal probability). The Rademacher complexity measure $\mathfrak R_n$ is
    \begin{equation}
        \mathfrak R_n(\cH) \coloneq \expv_{\varepsilon,D} [\sup_{f\in\cH} \frac{1}{n}\sum_{i=1}^n \varepsilon_i\ell(Y_i, f(X_i))] = \expv_{\varepsilon,D} \sup_{f\in\cH} \langle \varepsilon, \ell(f(X),Y)\rangle
    \end{equation}
    and it is a deterministic object depending on $n, \cH$, and the distributions of $\varepsilon$ and $D$. 
\end{definition}

% \begin{remark}
%     In our setting, where $\mathrm{Im}(f) = \{-1,1\}$, using $\mathbf{1}_{y_i f(x_i)}$ or directly $y_i f(x_i)$ is equivalent. 
% \end{remark}

\begin{remark}
    The Rademacher complexity measures the highest possible alignment between the function $(x,y)\mapsto \ell(f(x),y)$ and the Rademacher random variable $\varepsilon$. Since the loss function is given, it is a measure of the ``flexibility'' of $\cH$, hence its complexity.
\end{remark}

The Rademacher complexity controls the bias in the empirical risk $\hat R(f)$ through a symmetrization property (this is Proposition $4.2$ in~\cite{bach2024learning}). 

\begin{proposition}[Symmetrization]
    Consider data $D: \Omega^n \to \X^n \times \Y^n$ and an hypotheses set $\cH$. Given the Rademacher complexity $\mathfrak R_n$ as in Definition~\ref{def: rademacher}, the empirical and the expected risk of the estimators $f\in\cH$ satisfy the following bound
    \begin{equation}
        \expv_D[\sup_{f\in\cH} (R-\hat R_D) (f)] \le 2 \mathfrak R_n(\cH).
    \end{equation}
    \begin{proof}
        Let $h:(x,y)\mapsto(\ell(f(x),y))$ living in $\ell \circ \mathcal H$ and $Z\coloneqq(X,Y)$, so that
        \begin{equation*}
            R(f) =  \expv_D[h(Z)] \qquad \text{and} \qquad
            \hat R_D(f) = \frac{1}{n}\sum_{i} h(Z_i).
        \end{equation*}
        The key ingredient of the proof is to introduce an independent copy of our dataset $D$, $D'$, and study the random variable $h(Z') - h(Z)$, which has a symmetric distribution and hence is easier to analyze ({\it\/ symmetrization trick}). 
        We proceed as follows. 

        Introduce $D^\prime = {(Z_i')}_{i=1}^n$ identical copies of $Z'$ and rewrite
        \begin{equation}
            \expv_D[h(Z)] = \expv_{D^\prime\vert D} [h(Z_i')\vert D].
        \end{equation}
        Then,
        \begin{align}
            \expv_D[\sup_{f\in\cH} (R-\hat R) (f)] &= \expv_D\bigg[\sup_{h\in(\ell\circ\mathcal H)} \expv_{D^\prime\vert D} \Big[h(Z_i')\Big\vert D\Big] - \frac{1}{n}\sum_i h(Z_i)\bigg] \notag\\\notag
            &= \expv_D\bigg[\sup_{h\in(\ell\circ\mathcal H)} \frac{1}{n} \sum_i \expv_{D^\prime\vert D} \Big[h(Z_i')\Big\vert D\Big]-  h(Z_i) \bigg]\\
            &= \expv_D\bigg[\sup_{h\in(\ell\circ\mathcal H)}  \expv_{D^\prime\vert D} \Big[ \frac{1}{n} \sum_i h(Z_i')-  h(Z_i)\Big\vert D\Big] \bigg].
        \end{align} 
        Using that $f(X)\le \sup f(X) \Rightarrow \expv[f(X)]\le \expv[\sup f(X)] \Rightarrow \sup \expv[f(X)]\le \expv[\sup f(X)]$ and the law of total expectations, we obtain
        \begin{align}
            \expv_D[\sup_{f\in\cH} (R-\hat R) (f)] &\le \expv_D\bigg[ \expv_{D^\prime\vert D}\Big[ \sup_{h\in(\ell\circ\mathcal H)} \frac{1}{n} \sum_i h(Z_i')-  h(Z_i) \Big\vert D\Big]\bigg] \notag\\
            &=\expv_{D,D^\prime}\bigg[  \sup_{h\in(\ell\circ\mathcal H)} \frac{1}{n} \sum_i h(Z_i')-  h(Z_i) \bigg].\label{eq: 26}
        \end{align}
        Now we want to apply the symmetrization trick. 
        % parliamo direttamente di joint symmetry perché è questa che ci serve per concludere la cosa sul sup
        % la cit per questa parte è chat :((
        % quetsa proof si estende facilmente al caso H infinito perche si dimostra uguaglianza delle leggi per ogni collezione finita h1,...,hm e quindi si puo concludere (up to measurability or something) uguaglianza in legge dei due processi indexed by h\in\cH
        Let $m \coloneqq \vert\cH\vert$. For $i=1,\ldots,n$, the random variables $(h_1({Z_i}^\prime)-h_1(Z_i),\ldots, h_m({Z_i}^\prime)-h_m(Z_i))$ are jointly symmetric. 
        This is because $Z_i$ and ${Z_i}^\prime$ are identically distributed, hence 
        \begin{equation}
            {\big(h_j({Z_i}^\prime) -h_j(Z_i)\big)}_{j=1}^m \overset{d}{=} {\big(h_j(Z_i) - h_j({Z_i^\prime})\big)}_{j=1}^m.
        \end{equation}
        This means that for $e_i\in\{-1,1\}$,
        \begin{equation}
            {\big(h_j({Z_i}^\prime) -h_j(Z_i)\big)}_{j=1}^m \overset{d}{=} {\big(e_i (h_j({Z_i}^\prime) - h_j(Z_i))\big)}_{j=1}^m.
            %\qquad \forall i=1,\ldots,n, \forall h\in\ell\circ\cH
        \end{equation}
        Using independence over the $i$s we have 
        \begin{equation}
             {\bigg(\frac{1}{n}\sum_i h_j({Z_i}^\prime) -h_j(Z_i)\bigg)}_{j=1}^m \overset{d}{=} {\bigg(\frac{1}{n}\sum_i e_i(h_j({Z_i}^\prime) -h_j(Z_i))\bigg)}_{j=1}^m,
        \end{equation}
        and thus 
        \begin{equation}
            \sup_{h \in(\ell \circ\cH)} \frac{1}{n}\sum_i h({Z_i}^\prime) -h(Z_i) \overset{d}{=}\sup_{h \in(\ell\circ\cH)} \frac{1}{n}\sum_i e_i(h({Z_i}^\prime) -h(Z_i)),
        \end{equation}
        where since the supremum is a functional of the stochastic process, equality in distribution of the supremum follows from equality in distribution of the process.
        Hence, we obtain
        \begin{align}
            \expv_{D,D^\prime} \bigg[\sup_{h\in(\ell\circ\mathcal H)}\frac{1}{n} \sum_i h({Z_i}^\prime)-h(Z_i)\bigg] &= \expv_{D,D^\prime} \bigg[\sup_{h\in(\ell\circ\mathcal H)}\frac{1}{n}\sum_i e_i(h({Z_i}^\prime)-h(Z_i))\bigg]\\
            &= \expv_{D,D^\prime\vert\varepsilon} \bigg[\sup_{h\in(\ell\circ\mathcal H)}\frac{1}{n}\sum_i \varepsilon_i(h({Z_i}^\prime)-h(Z_i))\big\vert \varepsilon = e\bigg]\\
            &= \expv_{D,D^\prime,\varepsilon}\bigg[  \sup_{h\in(\ell\circ\mathcal H)} \frac{1}{n} \sum_i \varepsilon_i(h(Z_i')-  h(Z_i)) \bigg].
        \end{align}
        where in the last row we take the expected value over the disorder of $\varepsilon$ and use the tower law.
        Finally, inequality (\ref{eq: 26}) becomes
        \begin{align*}
            \expv_D[\sup_{f\in\cH} (R-\hat R) (f)] &\le \expv_{D,D^\prime,\varepsilon}\bigg[  \sup_{h\in(\ell\circ\mathcal H)} \frac{1}{n} \sum_i \varepsilon_i(h(Z_i')-  h(Z_i)) \bigg]\\
            &\le \expv_{D^\prime,\varepsilon}\bigg[  \sup_{h\in(\ell\circ\mathcal H)} \frac{1}{n} \sum_i \varepsilon_i h(Z_i') \bigg] + \expv_{D,\varepsilon}\bigg[ \sup_{h\in(\ell\circ\mathcal H)} \frac{1}{n} \sum_i - \varepsilon_i h(Z_i) \bigg]\\
            &= \expv_{D^\prime,\varepsilon}\bigg[  \sup_{h\in(\ell\circ\mathcal H)} \frac{1}{n} \sum_i \varepsilon_i h(Z_i') \bigg] + \expv_{D,\varepsilon}\bigg[ \sup_{h\in(\ell\circ\mathcal H)} \frac{1}{n} \sum_i \varepsilon_i h(Z_i) \bigg]\\
            &= 2 \mathfrak R_n(\cH)
        \end{align*}
        because $\varepsilon h(Z) \overset{d}{=} -\varepsilon h(Z)$.
    \end{proof}
\end{proposition}

The following result gives us an explicit bound on the Rademacher complexity. This is Theorem $3.3$ in~\cite{mohri2018foundations}. 

\begin{proposition}[Massart's lemma]
    Let $\ell:\Y\to\Y$ be as in Assumption~\ref{ass: lipsch loss}, so that $\ell(\cdot) \le M$ by Proposition~\ref{prop: suff assump 2.2}. Then, the Rademacher complexity of $\cH$ satisfies

    \begin{equation}
        \mathfrak R_n(\cH) = \expv_{\varepsilon,D} \bigg[\sup_{f\in\cH} \frac{1}{n}\sum_{i=1}^n \varepsilon_i\ell(Y_i, f(X_i))\bigg] \le M\sqrt{\frac{2\log \vert \cH \vert}{n}}.
    \end{equation}

    \begin{proof}
        For any $t>0$,
        \begin{align}
            \exp\bigg(t\, \expv_{\varepsilon,D} \Big[\sup_{f\in\cH}\sum_{i=1}^n \varepsilon_i \ell(Y_i,f(X_i))\Big]\bigg)
            &\le \expv_{\varepsilon,D}\bigg[\exp\Big(t\sup_{f\in\cH}\sum_{i=1}^n \varepsilon_i \ell(Y_i,f(X_i))\Big)\bigg]\notag\\
            &= \expv_{\varepsilon,D}\bigg[\sup_{f\in\cH}\exp\Big(t\sum_{i=1}^n \varepsilon_i \ell(Y_i,f(X_i))\Big)\bigg]\notag\\
            &\le \expv_{\varepsilon,D}\bigg[\sum_{f\in\cH}\exp\Big(t\sum_{i=1}^n \varepsilon_i \ell(Y_i,f(X_i))\Big)\bigg]\notag\\
            &= \sum_{f\in\cH}\prod_{i=1}^n \expv_{\varepsilon_i}\Big[\exp\big(t\varepsilon_i \ell(Y_i,f(X_i))\big)\Big].\label{eq: massart-start}
        \end{align}
        where we use Jensen's inequality on the convex function $\exp$ in the first inequality, we trivially bound the supremum with the sum, and we use the independence of the $\varepsilon_i$'s across $i=1,\ldots,n$ in the last equality. 

        Now, we want to apply Hoeffding's lemma (Theorem~\ref{app: hoeffdinglemma} in the appendix) to each factor. Notice that we can do so, because $\varepsilon_i \ell(Y_i,f(X_i))$ is a centered random variable ($\expv[\varepsilon_i]=0$ and $\varepsilon_i$ is independent of $\ell(Y_i,f(X_i))$) and it satisfies $\vert \varepsilon_i \ell(Y_i,f(X_i)) \vert \le M$ by assumption. We obtain
        \begin{equation}
            \expv_{\varepsilon_i}\Big[\exp\big(t\varepsilon_i \ell(Y_i,f(X_i))\big)\Big] \le e^{t^2M^2/2}
        \end{equation}
        and plugging this back into (\ref{eq: massart-start}), we obtain
        \begin{equation}\label{eq: massart-hoeffding}
            \exp\bigg(t\, \expv_{\varepsilon,D} \Big[\sup_{f\in\cH}\sum_{i=1}^n \varepsilon_i \ell(Y_i,f(X_i))\Big]\bigg) \le \sum_{f\in\cH}\prod_{i=1}^n e^{t^2M^2/2} = \vert \cH \vert e^{nt^2M^2/2},
        \end{equation}
        Now take the log of both sides
        \begin{equation}
            t\, \expv_{\varepsilon,D} \Big[\sup_{f\in\cH}\sum_{i=1}^n \varepsilon_i \ell(Y_i,f(X_i))\Big] \le \log \vert \cH \vert + \frac{nt^2M^2}{2}.
        \end{equation}
        and divide  by $tn>0$, obtaining
        \begin{equation}\label{eq: massart-opt}
            \mathfrak R_n(\cH) \le \frac{\log\vert \cH\vert}{tn} + \frac{tM^2}{2}.
        \end{equation}
        Since (\ref{eq: massart-opt}) holds for every $t>0$, we look for the tightest bound. We have
        \begin{equation}
            -\frac{\log\vert\cH\vert}{t^2n} + \frac{M^2}{2} = 0 \quad \Longrightarrow \quad t^\star = \sqrt{\frac{2\log\vert\cH\vert}{nM^2}},
        \end{equation}
        and
        \begin{equation}
            2\frac{\log\vert\cH\vert}{t^3 n} \ge 0
        \end{equation}
        hence $t^*$ is a minimizer since the RHS of (\ref{eq: massart-opt}). Plugging $t^\star$ back into (\ref{eq: massart-opt}) yields
        \begin{equation}
            \mathfrak R_n(\cH) \le M\sqrt{\frac{2\log\vert\cH\vert}{n}}.
        \end{equation}
  
    \end{proof}
\end{proposition}

Let

D=((X_1,Y_1),\ldots,(X_n,Y_n))

be the training sample. In the noiseless setting,

Y_i=t(X_i),

t:\mathcal X\to\{0,1\}.

Let $\mathcal H_d$ denote the class of one-dimensional decision trees of depth at most $d$, and let
\[
\ell:\{0,1\}\times\{0,1\}\to\{0,1\},
\qquad
\ell(\hat y,y)=\mathbf 1\{\hat y\neq y\},
\]
be the misclassification loss.

The empirical Rademacher complexity of the loss class is
\[
\widehat{\mathfrak R}_{D}(\ell\circ\mathcal H_d)
=
\frac1n
\mathbb E_\sigma
\left[
\sup_{h\in\mathcal H_d}
\sum_{i=1}^n
\sigma_i\,
\ell(h(X_i),Y_i)
\right].
\]

Since $Y_i=t(X_i)$, this can be written as
\[
\widehat{\mathfrak R}_{D}(\ell\circ\mathcal H_d)
=
\frac1n
\mathbb E_\sigma
\left[
\sup_{h\in\mathcal H_d}
\sum_{i=1}^n
\sigma_i\,
\ell(h(X_i),t(X_i))
\right].

{(\ell\circ\mathcal H_d)}_D(r)

\frac1n
\sum_{i=1}^n
\ell(h(X_i),Y_i)
\le r
\right\}.
\]
